-- 1. CREAR BASE DE DATOS (O USARLA SI YA EXISTE)
DROP DATABASE IF EXISTS veterinaria_almacen; 
CREATE DATABASE IF NOT EXISTS veterinaria_almacen;
USE veterinaria_almacen;

-- =======================================================
-- 9. TABLA DE ROLES (DEBE IR PRIMERO)
-- =======================================================
CREATE TABLE IF NOT EXISTS roles (
    id INT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
);

-- 2. TABLA DE CATEGORÍAS
CREATE TABLE IF NOT EXISTS Categoria (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL
);

-- 3. TABLA DE PROVEEDORES
CREATE TABLE IF NOT EXISTS Proveedor (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    ruc VARCHAR(20),
    telefono VARCHAR(20),
    direccion VARCHAR(150)
);

-- 4. TABLA DE USUARIOS (¡AÑADIDA RELACIÓN CLAVE!)
CREATE TABLE IF NOT EXISTS usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre_usuario VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    contraseña VARCHAR(64) NOT NULL,
    rol_id INT NOT NULL, 
    activo BOOLEAN DEFAULT TRUE,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    telefono VARCHAR(20) DEFAULT NULL,
    -- AÑADIDO: LA RELACIÓN FORMAL CON LA TABLA ROLES
    FOREIGN KEY (rol_id) REFERENCES roles(id) 
);

-- 5. TABLA DE PRODUCTOS (Incluye campos para RF17, RF34, RF36)
CREATE TABLE IF NOT EXISTS Producto (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    cantidad INT DEFAULT 0,
    precio DOUBLE NOT NULL,
    fecha_fabricacion DATE,
    fecha_caducidad DATE,
    stock_minimo INT DEFAULT 5, 
    estado VARCHAR(20) DEFAULT 'Activo', 
    fecha_ultima_modificacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    categoria_id INT,
    proveedor_id INT,
    FOREIGN KEY (categoria_id) REFERENCES Categoria(id),
    FOREIGN KEY (proveedor_id) REFERENCES Proveedor(id)
);

-- 6. TABLA DE ALMACÉN / HISTORIAL (Incluye motivo RF9)
CREATE TABLE IF NOT EXISTS almacen (
    id INT AUTO_INCREMENT PRIMARY KEY,
    producto_id INT NOT NULL,
    cantidad_movimiento INT NOT NULL,
    fecha_entrada TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    usuario_id INT NOT NULL,
    motivo VARCHAR(100) DEFAULT 'Venta',
    FOREIGN KEY (producto_id) REFERENCES Producto(id) ON DELETE CASCADE,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

-- 7. TABLA DE AUDITORÍA DE LOGIN (RF39)
CREATE TABLE IF NOT EXISTS historial_login (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    fecha_login TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ip_address VARCHAR(50),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
);

-- 8. TABLA DE TOKENS DE RECUPERACIÓN (RF30)
CREATE TABLE IF NOT EXISTS recuperacion_tokens (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT NOT NULL,
    token VARCHAR(64) NOT NULL,
    expiracion TIMESTAMP NOT NULL,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
);
